## Install

```npm
npx ondai add latency-chip
```

## Usage

```tsx
import { LatencyChip } from '@/components/ondai/latency-chip'

<LatencyChip latencyMs={metrics.latencyMs} tokensPerSecond={metrics.tokensPerSecond} provider="apple" />
```
