## Install

```bash
npx ondai add ion-border
```

## Usage

```tsx
import { IonBorder } from '@/components/ondai/ion-border'

<IonBorder active={isThinking} radius="lg" glow>
  <YourCard />
</IonBorder>
```

## Variants

| Prop | Options |
